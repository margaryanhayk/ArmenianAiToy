# Areg firmware as an ESP-IDF v5.5.4 project (arduino-esp32 3.3.8 as a
# component). Compiles the SAME sources as the Arduino sketch in
# ../AregVoiceMvp -- nothing is copied into the repo; the sketch is staged
# into the BUILD directory at configure time (exactly what arduino-cli does
# with its own build/sketch/ copy), and every staged file is a configure
# dependency, so editing the sketch re-stages it on the next build.
#
# Build with tools/firmware/build_idf.sh dev|release (it checks the IDF
# commit, fetches the pinned deps, and for release pads/gates/signs).
# Direct use (IDF_COMPONENT_MANAGER=0: never fetch unpinned components):
#   IDF_COMPONENT_MANAGER=0 idf.py -B build-dev -DAREG_PROFILE=dev -DSDKCONFIG=build-dev/sdkconfig build
#
# Inputs (cache variables / environment):
#   AREG_PROFILE            dev (default) | release
#   AREG_IDF_DEPS           directory filled by tools/firmware/idf_fetch_deps.py
#   AREG_FW_VERSION         release: required, MAJOR.MINOR.PATCH
#   AREG_FW_BUILD           informational build tag (default: "idf")
#   AREG_BACKEND_BASE_URL   release: required, https://...
#   env AREG_MANIFEST_HMAC_KEY  release: required. Read from the ENVIRONMENT
#                           at configure time, never stored in CMakeCache.
cmake_minimum_required(VERSION 3.16)

if(NOT AREG_PROFILE)
    set(AREG_PROFILE "dev")
endif()
if(NOT AREG_PROFILE STREQUAL "dev" AND NOT AREG_PROFILE STREQUAL "release")
    message(FATAL_ERROR "AREG_PROFILE must be dev or release (got '${AREG_PROFILE}')")
endif()
if(NOT AREG_IDF_DEPS)
    set(AREG_IDF_DEPS "$ENV{AREG_IDF_DEPS}")
endif()
if(NOT AREG_IDF_DEPS OR NOT EXISTS "${AREG_IDF_DEPS}/components/arduino/CMakeLists.txt")
    message(FATAL_ERROR "AREG_IDF_DEPS ('${AREG_IDF_DEPS}') has no fetched components -- run tools/firmware/idf_fetch_deps.py")
endif()
get_filename_component(AREG_IDF_DEPS "${AREG_IDF_DEPS}" ABSOLUTE)

# Base = arduino-esp32 3.3.8's own esp32s3 sdkconfig, then the Areg/FQBN
# overrides, then the profile (later file wins).
set(SDKCONFIG_DEFAULTS
    "${AREG_IDF_DEPS}/arduino-libs/sdkconfig"
    "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults"
    "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults.${AREG_PROFILE}")
set(EXTRA_COMPONENT_DIRS "${AREG_IDF_DEPS}/components")

# ---- stage the sketch (see header) -----------------------------------------
get_filename_component(AREG_SKETCH_DIR "${CMAKE_CURRENT_LIST_DIR}/../AregVoiceMvp" ABSOLUTE)
set(AREG_STAGE "${CMAKE_BINARY_DIR}/areg_sketch")
file(GLOB areg_sketch_files CONFIGURE_DEPENDS
     "${AREG_SKETCH_DIR}/*.ino" "${AREG_SKETCH_DIR}/*.cpp"
     "${AREG_SKETCH_DIR}/*.c" "${AREG_SKETCH_DIR}/*.h")
# configure_file(COPYONLY) rewrites a staged file only when its content
# changed, so an unchanged sketch does not trigger a rebuild.
set(areg_staged "config.h;AregVoiceMvp.ino.cpp;areg_build_defs.h")
foreach(f ${areg_sketch_files})
    get_filename_component(n "${f}" NAME)
    if(NOT n STREQUAL "config.h")
        configure_file("${f}" "${AREG_STAGE}/${n}" COPYONLY)
        list(APPEND areg_staged "${n}")
    endif()
endforeach()
# Drop staged files whose source was deleted from the sketch.
file(GLOB areg_stale "${AREG_STAGE}/*")
foreach(f ${areg_stale})
    get_filename_component(n "${f}" NAME)
    if(NOT n IN_LIST areg_staged)
        file(REMOVE "${f}")
    endif()
endforeach()
function(areg_write_if_changed path content)
    file(WRITE "${path}.tmp" "${content}")
    configure_file("${path}.tmp" "${path}" COPYONLY)
    file(REMOVE "${path}.tmp")
endfunction()
# config.h: a RELEASE image is built from config.h.example ONLY (never a
# developer's local config.h with bench credentials in it); release values
# arrive through areg_build_defs.h below. DEV uses the local config.h when
# there is one, like the Arduino IDE does.
if(AREG_PROFILE STREQUAL "dev" AND EXISTS "${AREG_SKETCH_DIR}/config.h")
    configure_file("${AREG_SKETCH_DIR}/config.h" "${AREG_STAGE}/config.h" COPYONLY)
else()
    configure_file("${AREG_SKETCH_DIR}/config.h.example" "${AREG_STAGE}/config.h" COPYONLY)
endif()
# The .ino as a translation unit: what arduino-builder does (the sketch
# already declares everything it uses, so no generated prototypes needed).
areg_write_if_changed("${AREG_STAGE}/AregVoiceMvp.ino.cpp"
     "#include <Arduino.h>\n#include \"AregVoiceMvp.ino\"\n")

# Force-included into every sketch translation unit (main component only).
set(defs "// GENERATED by esp32/AregVoiceIdf/CMakeLists.txt -- do not edit.\n")
if(AREG_PROFILE STREQUAL "release")
    if(NOT AREG_FW_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
        message(FATAL_ERROR "release: AREG_FW_VERSION must be MAJOR.MINOR.PATCH (got '${AREG_FW_VERSION}')")
    endif()
    if(NOT AREG_BACKEND_BASE_URL MATCHES "^https://[^\"\\\\ ]+$")
        message(FATAL_ERROR "release: AREG_BACKEND_BASE_URL must be an https:// URL")
    endif()
    set(hmac "$ENV{AREG_MANIFEST_HMAC_KEY}")
    if(hmac STREQUAL "" OR hmac MATCHES "[\"\\\\]")
        message(FATAL_ERROR "release: set AREG_MANIFEST_HMAC_KEY in the environment (no quotes/backslashes)")
    endif()
    string(APPEND defs "#define AREG_SECURITY_PROFILE_RELEASE 1\n")
    string(APPEND defs "#define AREG_FW_VERSION \"${AREG_FW_VERSION}\"\n")
    string(APPEND defs "#define AREG_BACKEND_BASE_URL \"${AREG_BACKEND_BASE_URL}\"\n")
    string(APPEND defs "#define AREG_MANIFEST_HMAC_KEY \"${hmac}\"\n")
    # config.h.example defines AREG_CONTENT_SYNC_BENCH only inside its
    # `#ifndef AREG_FW_VERSION` block; defining the version here would
    # silently drop content sync (the "BENCH" name lies -- it is load-bearing).
    string(APPEND defs "#define AREG_CONTENT_SYNC_BENCH 1\n")
else()
    string(APPEND defs "#define AREG_SECURITY_PROFILE_DEV 1\n")
    if(AREG_FW_VERSION)
        string(APPEND defs "#define AREG_FW_VERSION \"${AREG_FW_VERSION}\"\n#define AREG_CONTENT_SYNC_BENCH 1\n")
    endif()
endif()
if(NOT AREG_FW_BUILD)
    set(AREG_FW_BUILD "idf")
endif()
string(APPEND defs "#define AREG_FW_BUILD \"${AREG_FW_BUILD}\"\n")
areg_write_if_changed("${AREG_STAGE}/areg_build_defs.h" "${defs}")
set(hmac "")

if(AREG_FW_VERSION)
    set(PROJECT_VER "${AREG_FW_VERSION}")  # esp_app_desc: no `git describe`
else()
    set(PROJECT_VER "dev")
endif()

include($ENV{IDF_PATH}/tools/cmake/project.cmake)

# Board defines arduino-cli passes for
# esp32:esp32:esp32s3:PSRAM=opi,FlashSize=8M,PartitionScheme=custom,CDCOnBoot=cdc
# (USBMode default hwcdc, DebugLevel none).
idf_build_set_property(COMPILE_DEFINITIONS "ARDUINO_USB_MODE=1" APPEND)
idf_build_set_property(COMPILE_DEFINITIONS "ARDUINO_USB_CDC_ON_BOOT=1" APPEND)
idf_build_set_property(COMPILE_DEFINITIONS "ARDUINO_USB_MSC_ON_BOOT=0" APPEND)
idf_build_set_property(COMPILE_DEFINITIONS "ARDUINO_USB_DFU_ON_BOOT=0" APPEND)
idf_build_set_property(COMPILE_DEFINITIONS "BOARD_HAS_PSRAM" APPEND)
idf_build_set_property(COMPILE_DEFINITIONS "CORE_DEBUG_LEVEL=0" APPEND)
idf_build_set_property(COMPILE_DEFINITIONS "F_CPU=240000000L" APPEND)
# Reproducible __FILE__ strings for code outside the project and IDF trees
# (IDF itself maps the project dir to "." and IDF_PATH to /IDF).
idf_build_set_property(COMPILE_OPTIONS "-fmacro-prefix-map=${AREG_IDF_DEPS}=/AREG_DEPS" APPEND)
idf_build_set_property(COMPILE_OPTIONS "-fmacro-prefix-map=${CMAKE_BINARY_DIR}=/AREG_BUILD" APPEND)

project(AregVoiceMvp)
